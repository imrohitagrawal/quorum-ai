"""Run extract_one.py in a child, sample its RSS every 20 ms, kill at TIMEOUT s or RSS_CAP bytes."""
import json, subprocess, sys, time, os, glob
TIMEOUT=float(os.environ.get('TIMEOUT','20')); RSS_CAP=int(os.environ.get('RSS_CAP_MB','3000'))*2**20
PY=os.path.join(os.path.dirname(__file__),'.venv/bin/python'); EX=os.path.join(os.path.dirname(__file__),'extract_one.py')
def rss(pid):
    o=subprocess.run(['ps','-o','rss=','-p',str(pid)],capture_output=True,text=True).stdout.strip()
    return int(o)*1024 if o else 0
libs=sys.argv[1].split(','); files=sys.argv[2:]
for f in files:
  for lib in libs:
    t0=time.perf_counter(); p=subprocess.Popen([PY,EX,lib,f,'10'],stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    peak=0; killed=None
    while p.poll() is None:
        r=rss(p.pid); peak=max(peak,r)
        el=time.perf_counter()-t0
        if el>TIMEOUT: killed=f'TIMEOUT>{TIMEOUT}s'; p.kill(); break
        if r>RSS_CAP: killed=f'RSS>{RSS_CAP>>20}MB'; p.kill(); break
        time.sleep(0.02)
    out,err=p.communicate(); el=time.perf_counter()-t0
    res=None
    for line in out.splitlines():
        if line.startswith('{'): res=json.loads(line)
    errl=[l for l in err.splitlines() if l.strip()]
    print(json.dumps({'file':os.path.basename(f),'lib':lib,'killed':killed,'wall_total_s':round(el,2),
      'peak_rss_mb_sampled':round(peak/2**20),'child_rc':p.returncode,
      'chars':res and res['chars'],'npages':res and res['npages'],'err':res and res['err'],
      'sample':res and res['sample'][:60],'stderr_tail':(errl[-1][:150] if errl else None),'stderr_lines':len(errl)}),flush=True)
